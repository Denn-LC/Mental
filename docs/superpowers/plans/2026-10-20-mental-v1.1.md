**Goal:** Replace hints with Give up, add arrow-key menus with a 3-2-1 countdown and Free Play buttons, make the test 2 minutes of mixed topics and levels with no wrong-answer feedback, stop pointless 0/1 questions, use the pixel font everywhere with closeness-based wrong-answer lines, and fix the bugs deferred from v1.0.

**Architecture:** Same shape as v1.0: pure-Python engine (`src/mental/engine/`), thin FastAPI app, one static page. The engine loses hints and gains number rules; the attempt record swaps `hints_used` for `gave_up`; `make_questions` takes a list of levels so the test can mix them; the page gets home/options screens, a countdown, and a session counter that makes stale fetches harmless.

## Global Constraints

- Runtime deps stay `fastapi`, `uvicorn`; dev deps stay `pytest`, `httpx2`. Playwright is never added to `pyproject.toml`.
- `src/mental/engine/` never imports FastAPI or `mental.web`.
- Every answer is an `int` with `|answer| ≤ 99_999`; every step is a true `"<sum> = <result>"`; the last step's result is the answer.
- Display minus is ASCII `-` (the pixel font has no `−`). The answer box accepts both.
- No 0 or 1 in question text or on the left of a step, except `near_round_add`/`near_round_subtract` steps.
- Round second numbers in basic add/subtract only at level 1.
- No visual design work beyond what the spec lists (pixel font everywhere, plain menu/button styling).
- Every commit message ends with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.

```python
"""Question generators.

Each strategy is a function (rng, difficulty) -> (text, answer, steps).
It picks numbers that suit one mental trick and writes the worked method for them.
Every step must be a true sum written as "<sum> = <result>" (the tests check this).
"""


def fmt(n):
    """1234 -> '1,234'. Minus stays a plain "-": the pixel font has no "-" glyph."""
    return f"{n:,}"


def random_number(rng, digits):
    return rng.randint(10 ** (digits - 1), 10**digits - 1)


def place_parts(n):
    """2917 -> [2000, 900, 10, 7]"""
    s = str(n)
    return [int(d) * 10 ** (len(s) - 1 - i) for i, d in enumerate(s) if d != "0"]


def chunk_steps(start, parts, subtract=False):
    """Add (or take away) parts one at a time: ['4,382 + 2,000 = 6,382', ...]"""
    steps = []
    total = start
    for part in parts:
        new = total - part if subtract else total + part
        op = "-" if subtract else "+"
        steps.append(f"{fmt(total)} {op} {fmt(part)} = {fmt(new)}")
        total = new
    return steps


# --- addition -------------------------------------------------------------


def basic_add(rng, difficulty):
    digits = difficulty + 1  # level 1: 2 digits, 2: 3 digits, 3: 4 digits
    a, b = random_number(rng, digits), random_number(rng, digits)
    b += 0 if b % 10 else 1  # a round b would make the method one step
    return f"{fmt(a)} + {fmt(b)}", a + b, chunk_steps(a, place_parts(b))


def near_round(rng, difficulty):
    """A number just under a round one, e.g. 197 = 200 - 3. Returns (round, off)."""
    if difficulty == 2:
        return rng.choice(range(100, 1000, 100)), rng.randint(1, 4)
    return rng.choice(range(100, 10000, 100)), rng.randint(1, 9)


def near_round_add(rng, difficulty):
    round_, off = near_round(rng, difficulty)
    b = round_ - off
    a = random_number(rng, difficulty + 1)
    steps = [
        f"{fmt(a)} + {fmt(round_)} = {fmt(a + round_)}",
        f"{fmt(a + round_)} - {off} = {fmt(a + b)}",
    ]
    return f"{fmt(a)} + {fmt(b)}", a + b, steps


def signed_add(rng, difficulty):
    x, y = random_number(rng, rng.choice([3, 4])), random_number(rng, rng.choice([3, 4]))
    answer = y - x
    big, small = max(x, y), min(x, y)
    steps = [
        f"{fmt(big)} - {fmt(small)} = {fmt(big - small)}",
        f"{fmt(-x)} + {fmt(y)} = {fmt(answer)}",
    ]
    return f"{fmt(-x)} + {fmt(y)}", answer, steps


# --- subtraction ----------------------------------------------------------


def basic_subtract(rng, difficulty):
    digits = difficulty + 1
    b = rng.randint(10 ** (digits - 1), 10**digits - 2)  # leaves room for a > b
    b += 0 if b % 10 else 1  # a round b would make the method one step
    a = rng.randint(b + 1, 10**digits - 1)  # never x - x
    return f"{fmt(a)} - {fmt(b)}", a - b, chunk_steps(a, place_parts(b), subtract=True)


def near_round_subtract(rng, difficulty):
    round_, off = near_round(rng, difficulty)
    b = round_ - off
    a = rng.randint(b + 1, 10 ** (difficulty + 1) - 1)
    steps = [
        f"{fmt(a)} - {fmt(round_)} = {fmt(a - round_)}",
        f"{fmt(a - round_)} + {off} = {fmt(a - b)}",
    ]
    return f"{fmt(a)} - {fmt(b)}", a - b, steps


def signed_subtract(rng, difficulty):
    x, y = random_number(rng, rng.choice([3, 4])), random_number(rng, rng.choice([3, 4]))
    answer = -(x + y)
    steps = [
        f"{fmt(x)} + {fmt(y)} = {fmt(x + y)}",
        f"{fmt(-x)} - {fmt(y)} = {fmt(answer)}",
    ]
    return f"{fmt(-x)} - {fmt(y)}", answer, steps


# --- multiplication -------------------------------------------------------


def basic_multiply(rng, difficulty):
    if difficulty == 1:
        a, b = rng.randint(2, 9), rng.randint(2, 9)
        steps = [f"{a} × {b - 1} = {a * (b - 1)}", f"{a * (b - 1)} + {a} = {a * b}"]
        return f"{a} × {b}", a * b, steps
    a = rng.choice([n for n in range(12, 100) if n % 10])
    b = rng.randint(3, 9)
    tens, units = a - a % 10, a % 10
    steps = [
        f"{tens} × {b} = {tens * b}",
        f"{units} × {b} = {units * b}",
        f"{tens * b} + {units * b} = {a * b}",
    ]
    return f"{a} × {b}", a * b, steps


def near_round_multiply(rng, difficulty):
    if difficulty == 2:
        n, m = rng.choice([9, 11, 19, 21]), rng.randint(12, 49)
    else:
        n, m = rng.choice([29, 31, 39, 41, 49, 51, 99, 101]), rng.randint(12, 99)
    round_ = 100 if n > 90 else (n + 1) // 10 * 10
    sign = "+" if n > round_ else "-"
    base = round_ * m
    steps = [f"{round_} × {m} = {fmt(base)}", f"{fmt(base)} {sign} {m} = {fmt(n * m)}"]
    return f"{n} × {m}", n * m, steps


# --- division -------------------------------------------------------------


def basic_divide(rng, difficulty):
    if difficulty == 1:
        b, q = rng.randint(2, 9), rng.randint(2, 9)
        a = b * q
        return f"{a} ÷ {b}", q, [f"{b} × {q} = {a}", f"{a} ÷ {b} = {q}"]
    b = rng.randint(3, 9)
    q = rng.choice([n for n in range(12, 31) if n % 10])
    a = b * q
    tens = q - q % 10
    first, rest = b * tens, a - b * tens
    steps = [
        f"{first} ÷ {b} = {tens}",
        f"{rest} ÷ {b} = {q - tens}",
        f"{tens} + {q - tens} = {q}",
    ]
    return f"{a} ÷ {b}", q, steps


def divide_by_5(rng, difficulty):
    q = rng.randint(12, 400)
    a = 5 * q
    steps = [f"{fmt(a)} × 2 = {fmt(2 * a)}", f"{fmt(2 * a)} ÷ 10 = {fmt(q)}"]
    return f"{fmt(a)} ÷ 5", q, steps


def divide_by_25(rng, difficulty):
    q = rng.randint(12, 400)
    a = 25 * q
    steps = [f"{fmt(a)} × 4 = {fmt(4 * a)}", f"{fmt(4 * a)} ÷ 100 = {fmt(q)}"]
    return f"{fmt(a)} ÷ 25", q, steps


# --- percentages ----------------------------------------------------------


def percent_10(rng, difficulty):
    k = rng.randint(2, 99)
    base = 10 * k
    return f"10% of {fmt(base)}", k, [f"{fmt(base)} ÷ 10 = {k}"]


def percent_25(rng, difficulty):
    k = rng.randint(5, 250)
    base = 4 * k
    steps = [f"{fmt(base)} ÷ 2 = {fmt(2 * k)}", f"{fmt(2 * k)} ÷ 2 = {fmt(k)}"]
    return f"25% of {fmt(base)}", k, steps


def percent_15(rng, difficulty):
    k = rng.randint(2, 99)
    base = 20 * k  # a multiple of 20 keeps 5% a whole number
    ten, five = 2 * k, k
    steps = [f"{fmt(base)} ÷ 10 = {ten}", f"{ten} ÷ 2 = {five}", f"{ten} + {five} = {ten + five}"]
    return f"15% of {fmt(base)}", ten + five, steps


# --- relationships --------------------------------------------------------


def double_halve(rng, difficulty):
    if difficulty == 3:
        a, b = rng.randint(4, 30), 3 * rng.randint(2, 15)
        c, d = a * 3, b // 3
        steps = [f"{a} × 3 = {c}", f"{b} ÷ 3 = {d}"]
    else:
        a = 2 * rng.randint(2, 20 if difficulty == 1 else 49)
        b = rng.randint(3, 15 if difficulty == 1 else 30)
        c, d = a // 2, b * 2
        steps = [f"{a} ÷ 2 = {c}", f"{b} × 2 = {d}"]
    return f"{a} × {b} = {c} × ?", d, steps


def balance_sum(rng, difficulty):
    unit = 10 if difficulty == 2 else 100
    a = rng.choice([n for n in range(unit + 1, unit * 10) if n % unit])
    b = rng.randint(unit, unit * 10 - 1)
    c = (a // unit + 1) * unit
    move = c - a
    steps = [f"{fmt(c)} - {a} = {move}", f"{b} - {move} = {b - move}"]
    return f"{a} + {b} = {fmt(c)} + ?", b - move, steps
```

```python
"""Turn a list of attempt records into accuracy and speed numbers.

"accuracy" counts clean answers: correct on the first try.
"avg_time_ms" is over solved questions only, so skips and give-ups don't drag it down.
"""


def stats(attempts):
    solved = [a for a in attempts if a["correct"]]
    clean = [a for a in solved if not a["wrong_tries"]]
    return {
        "count": len(attempts),
        "score": len(solved),
        "skipped": sum(1 for a in attempts if a["skipped"]),
        "gave_up": sum(1 for a in attempts if a["gave_up"]),
        "accuracy": len(clean) / len(attempts) if attempts else 0.0,
        "avg_time_ms": round(sum(a["time_ms"] for a in solved) / len(solved)) if solved else 0,
    }


def group_stats(attempts, key):
    groups = {}
    for a in attempts:
        groups.setdefault(a[key], []).append(a)
    return {name: stats(group) for name, group in groups.items()}


def summarise(attempts):
    slowest = sorted(attempts, key=lambda a: a["time_ms"], reverse=True)[:5]
    return {
        "overall": stats(attempts),
        "by_category": group_stats(attempts, "category"),
        "by_strategy": group_stats(attempts, "strategy"),
        "slowest": [a["question_id"] for a in slowest],
    }
```

```python
def numbers_in(expr):
    """'1,234 + 17 = 1,251' -> [1234, 17, 1251]"""
    return [int(n.replace(",", "")) for n in re.findall(r"\d[\d,]*", expr)]


# "467 + 199 → 467 + 200 − 1": taking the 1 back off is the trick itself, not a pointless step
ONE_IS_THE_TRICK = {"near_round_add", "near_round_subtract"}


@pytest.mark.parametrize("strategy_id, difficulty", CASES)
def test_no_pointless_zero_or_one(strategy_id, difficulty):
    make = STRATEGIES[strategy_id]["make"]
    for seed in range(300):
        text, answer, steps = make(random.Random(seed), difficulty)
        assert not {0, 1} & set(numbers_in(text.replace("?", ""))), text
        if strategy_id not in ONE_IS_THE_TRICK:
            for step in steps:
                left = step.split(" = ")[0]
                assert not {0, 1} & set(numbers_in(left)), f"{text}: {step}"
```

```python
def make_questions(categories, difficulties, count, seed=None):
    """count questions, each from a random (category, level) pair that has questions."""
    pairs = [(c, d) for c in categories for d in difficulties if strategies_for(c, d)]

    return [make_question(*rng.choice(pairs), rng.randrange(2**32)) for _ in range(count)]
```

```js
// Questions arrive from /api/questions already carrying their answer and worked steps.

const TEST_SECONDS = 120;
const COUNTDOWN_MS = 700;  // per number of the 3-2-1 countdown
const QUESTIONS_PER_FETCH = { free: 20, test: 200 };
const KEYS = {
  free: "Enter submit / next · Tab give up · Esc menu",
  test: "type the answer · Tab skip · Esc menu",
};

const $ = (id) => document.getElementById(id);
const input = $("answer");

const state = {
  mode: "free",       // "free" | "test"
  category: "mixed",  // for /api/questions: "mixed" or a comma-separated list
  difficulty: "1",    // "1", "2", "3" or "mixed" (each question gets a random level)
  session: 0,         // goes up on every new game or return to the menu; stale fetches check it
  loading: false,     // a question fetch is in flight
  queue: [],          // fetched questions not shown yet
  seen: {},           // question id -> question, for the results screen
  current: null,      // question on screen (null while counting down, loading or between tests)
  countingDown: false,
  attempt: null,      // attempt record for the current question
  shownAt: 0,
  solved: false,      // free play: answered or given up, waiting for Enter
  streak: 0,
  attempts: [],
  testEndsAt: 0,      // 0 = no test running
  endedAt: -Infinity, // when the last test ended
  timer: null,
};

// --- menus -----------------------------------------------------------------

function showScreen(id) {
  for (const screen of document.querySelectorAll(".screen")) screen.hidden = screen.id !== id;
}

function showHome() {
  stopTest();
  state.session++;  // anything still loading for the old game is dropped
  state.loading = false;
  state.countingDown = false;
  state.current = null;
  showScreen("home");
  document.querySelector(`#home [data-mode=${state.mode}]`).focus();
}

function showOptions() {
  showScreen("options");
  $("menu-error").textContent = "";
  $("play").focus();
}

function chooseMode(mode) {
  state.mode = mode;
  if (mode === "free") return showOptions();
  state.category = "mixed";
  state.difficulty = "mixed";
  startGame();
}

function playFree() {
  const ticked = [...document.querySelectorAll("input[name=topic]:checked")].map((box) => box.value);
  if (ticked.length === 0) {
    $("menu-error").textContent = "Tick at least one topic.";
    return;
  }
  state.category = ticked.length === 6 ? "mixed" : ticked.join(",");
  state.difficulty = document.querySelector("input[name=level]:checked").value;
  startGame();
}

// Arrow keys move the highlight the way they point; Enter or Space picks the item.
function onMenuKey(event) {
  const direction = { ArrowUp: [0, -1], ArrowDown: [0, 1], ArrowLeft: [-1, 0], ArrowRight: [1, 0] }[event.key];
  if (direction) {
    nearestItem(document.activeElement, direction)?.focus();
  } else if (event.key === "Enter" && document.activeElement.tagName === "INPUT") {
    document.activeElement.click();  // tick a box or pick a level (buttons click on Enter by themselves)
  } else if (event.key === "Escape" && !$("options").hidden) {
    showHome();
  } else {
    return;
  }
  event.preventDefault();
}

// The menu item nearest to `from` in direction [dx, dy]. An item counts only if it lies more in
// that direction than off to the side. Null at the edge, so the highlight stays put.
function nearestItem(from, [dx, dy]) {
  const items = [...document.querySelectorAll(".screen:not([hidden]) [data-nav]")];
  if (!items.includes(from)) return items[0];
  const box = (el) => (el.closest("label") || el).getBoundingClientRect();  // a tickbox's label is what you see
  const gap = (v, low, high) => Math.max(low - v, 0, v - high);  // 0 when v is within [low, high]
  const here = box(from);
  const x = here.left + here.width / 2;
  const y = here.top + here.height / 2;
  let best = null;
  let bestScore = Infinity;
  for (const item of items) {
    const b = box(item);
    const ahead = (b.left + b.width / 2 - x) * dx + (b.top + b.height / 2 - y) * dy;
    const aside = dx ? gap(y, b.top, b.bottom) : gap(x, b.left, b.right);
    const score = ahead + aside * 3;
    if (ahead > 1 && aside < ahead && score < bestScore) {
      best = item;
      bestScore = score;
    }
  }
  return best;
}

function startGame() {
  showScreen("game");
  renderSettings();
  startRound();
}

// A round: 3-2-1 countdown while the questions load, then go (and start the clock in test mode).
async function startRound() {
  state.session++;
  const session = state.session;
  stopTest();
  state.loading = false;
  state.countingDown = true;
  state.current = null;
  state.queue = [];
  state.attempts = [];
  state.seen = {};
  setStreak(0);
  $("score").textContent = "score 0";
  $("timer").textContent = clock(TEST_SECONDS);
  $("results").hidden = true;
  input.value = "";
  showSteps([]);
  setFeedback("");
  renderButtons();
  showAnswerBox(false);
  $("game").classList.add("counting");
  const fetching = fetchQuestions(QUESTIONS_PER_FETCH[state.mode]).catch((error) => error);
  for (const n of [3, 2, 1]) {
    $("question").textContent = n;
    await new Promise((done) => setTimeout(done, COUNTDOWN_MS));
    if (session !== state.session) return;  // Esc during the countdown
  }
  const questions = await fetching;
  if (session !== state.session) return;
  state.countingDown = false;
  $("game").classList.remove("counting");
  if (questions instanceof Error) {
    $("question").textContent = "";
    setFeedback(`Could not load questions (${questions.message}). Enter to retry.`, "wrong");
    return;
  }
  state.queue = questions;
  showQuestion(state.queue.shift());
  if (state.mode === "test") {
    state.testEndsAt = performance.now() + TEST_SECONDS * 1000;
    state.timer = setInterval(tick, 100);
  }
}

// --- questions -------------------------------------------------------------

async function fetchQuestions(count) {
  const params = new URLSearchParams({ category: state.category, difficulty: state.difficulty, count });
  const response = await fetch(`/api/questions?${params}`);
  if (!response.ok) throw new Error(`server said ${response.status}`);
  return response.json();
}

async function nextQuestion() {
  if (state.queue.length === 0) {
    if (state.loading) return;  // already fetching
    const session = state.session;
    state.loading = true;
    state.current = null;  // nothing on screen to answer or skip while loading
    state.solved = false;
    try {
      const questions = await fetchQuestions(QUESTIONS_PER_FETCH[state.mode]);
      if (session !== state.session) return;  // went back to the menu while loading
      state.queue = questions;
    } catch (error) {
      if (session === state.session) setFeedback(`Could not load questions (${error.message}). Enter to retry.`, "wrong");
      return;
    } finally {
      if (session === state.session) state.loading = false;
    }
  }
  showQuestion(state.queue.shift());
}

function showQuestion(q) {
  state.current = q;
  state.seen[q.id] = q;
  state.solved = false;
  state.attempt = {
    question_id: q.id, seed: q.seed, text: q.text, category: q.category, strategy: q.strategy,
    difficulty: q.difficulty, answer: q.answer, given: "", correct: false, skipped: false,
    wrong_tries: [], gave_up: false, time_ms: 0, mode: state.mode,
  };
  state.shownAt = performance.now();
  $("question").textContent = q.text;
  showSteps([]);
  setFeedback("");
  renderButtons();
  showAnswerBox(true);
  input.value = "";
  input.focus();
}

// "−465" and "-465" both mean -465. Anything that isn't a whole number gives null.
function parseAnswer(text) {
  const clean = text.replace("−", "-");
  return /^-?\d+$/.test(clean) ? Number(clean) : null;
}

function finishAttempt(correct, skipped = false) {
  const a = state.attempt;
  a.given = input.value;
  a.correct = correct;
  a.skipped = skipped;
  a.time_ms = Math.round(performance.now() - state.shownAt);
  state.attempts.push(a);
  setStreak(correct && a.wrong_tries.length === 0 ? state.streak + 1 : 0);
  $("score").textContent = `score ${state.attempts.filter((x) => x.correct).length}`;
}

function submit() {
  const value = parseAnswer(input.value);
  if (value === null) return;
  if (value === state.current.answer) {
    finishAttempt(true);
    state.solved = true;
    setFeedback("correct!  Enter for the next one", "right");
    showSteps(state.current.steps);
    renderButtons();
    return;
  }
  state.attempt.wrong_tries.push(input.value);
  setFeedback(wrongLine(value, state.current.answer), "wrong");
  input.select();
}

// What to say after a wrong answer depends on how far off it was.
const WRONG_LINES = {
  close: ["almost!", "so close", "nearly there", "off by a hair"],
  off: ["not quite", "nope, try again", "not it", "try again"],
  way: ["are you even trying?", "jesus.", "that's not even close", "...what?", "bold guess"],
};
let lastWrongLine = "";

function wrongLine(given, answer) {
  const diff = Math.abs(given - answer);
  const gap = diff / Math.max(Math.abs(answer), 10);  // how far off, as a share of the answer
  const wrongSign = given * answer < 0;
  const tier = wrongSign || gap > 0.5 ? "way" : diff <= 1 || gap <= 0.05 ? "close" : "off";
  const options = WRONG_LINES[tier].filter((line) => line !== lastWrongLine);
  lastWrongLine = options[Math.floor(Math.random() * options.length)];
  return lastWrongLine;
}

function giveUp() {
  if (!state.current || state.solved) return;
  state.attempt.gave_up = true;
  finishAttempt(false);
  state.solved = true;
  const answer = state.current.steps.at(-1).split(" = ")[1];  // already formatted, e.g. "1,480"
  setFeedback(`the answer is ${answer}.  Enter for the next one`, "wrong");
  showSteps(state.current.steps);
  renderButtons();
}

function skip() {
  if (!state.current || !state.testEndsAt) return;  // only during a running test
  finishAttempt(false, true);
  nextQuestion();
}

function onInput() {
  // allow one leading minus and up to 5 digits (no answer is longer)
  const cleaned = input.value.replace(/[^\d\-−]/g, "").replace(/(?!^)[-−]/g, "").replace(/^([-−]?\d{0,5}).*$/, "$1");
  if (cleaned !== input.value) input.value = cleaned;
  // test mode accepts the right answer as soon as it is typed
  if (state.mode === "test" && state.testEndsAt && state.current &&
      parseAnswer(input.value) === state.current.answer) {
    finishAttempt(true);
    nextQuestion();
  }
}

function onEnter() {
  if (state.countingDown) return;
  if (state.mode === "test") {
    if (state.testEndsAt) return recordWrongTry();
    // between rounds: a late Enter from the last answer must not skip straight past the results
    if (performance.now() - state.endedAt > 1000) startRound();
    return;
  }
  if (!state.current || state.solved) return nextQuestion();  // also retries a failed load
  submit();
}

// Test mode gives no feedback on a wrong answer (that would be a hint), but keeps it for the stats.
function recordWrongTry() {
  const value = parseAnswer(input.value);
  if (state.current && value !== null && value !== state.current.answer) state.attempt.wrong_tries.push(input.value);
}

// --- test mode -------------------------------------------------------------

const clock = (seconds) => `${Math.floor(seconds / 60)}:${String(seconds % 60).padStart(2, "0")}`;

function tick() {
  const left = state.testEndsAt - performance.now();
  $("timer").textContent = clock(Math.max(0, Math.ceil(left / 1000)));
  if (left <= 0) endTest();
}

function stopTest() {
  clearInterval(state.timer);
  state.testEndsAt = 0;
}

async function endTest() {
  stopTest();
  const session = state.session;
  state.current = null;  // the unfinished question is not recorded
  state.endedAt = performance.now();
  $("question").textContent = "time!";
  input.value = "";
  showAnswerBox(false);
  let summary;
  try {
    const response = await fetch("/api/summary", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(state.attempts),

    if (!response.ok) throw new Error(`server said ${response.status}`);
    summary = await response.json();
  } catch (error) {
    if (session === state.session) setFeedback(`Could not load results (${error.message})`, "wrong");
    return;
  }
  if (session !== state.session) return;  // moved on while results loaded
  showResults(summary);
}

const percent = (x) => `${Math.round(x * 100)}%`;
const seconds = (ms) => `${(ms / 1000).toFixed(1)}s`;

    <p>Enter to go again · Esc for the menu</p>`;
  $("results").hidden = false;
}
```
